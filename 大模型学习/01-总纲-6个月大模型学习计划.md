## 1. 概述

### 1.1 学习路径总览

本指南将 6 个月学习周期拆分为 7 个阶段，合计约 480 小时。每个阶段有明确的目标、技术栈、时间预算和交付物。

|------|------|------|------|---------|

| 时段 | 时长 | 用途 | 说明 |

```python
# 装饰器实战
import time
from functools import wraps

def timer(func):
    """计时装饰器——记录函数执行时间"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"{func.__name__} 执行耗时: {elapsed:.4f}s")
        return result
    return wrapper

def retry(max_attempts=3, delay=1):
    """重试装饰器——失败自动重试"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        raise
                    print(f"重试 {attempt + 1}/{max_attempts}: {e}")
                    time.sleep(delay)
        return wrapper
    return decorator
```

```python
def clean_pipeline(df: pd.DataFrame) -> pd.DataFrame:
    """标准数据清洗流水线"""
    return (df
        .drop_duplicates()                       # 去重
        .dropna(subset=['text'])                 # 删除关键列为空的行
        .assign(
            text=lambda x: x['text'].str.strip(),   # 去除首尾空格
            length=lambda x: x['text'].str.len()    # 添加文本长度特征
        )
        .query('length >= 10')                   # 过滤过短文本
        .reset_index(drop=True)                  # 重置索引
    )

    'length': ['mean', 'std', 'count'],
    'label': 'first'
})
```

```python
class PromptExperiment:
    """Prompt 策略对比实验"""

    def __init__(self, client, task: str, test_cases: list[dict]):
        self.client = client
        self.task = task
        self.test_cases = test_cases
        self.results = []

    def run_strategy(self, name: str, template_fn):
        """运行一种 Prompt 策略并记录结果"""
        for case in self.test_cases:
            prompt = template_fn(case['input'])
            response = self.client.chat([{"role": "user", "content": prompt}])
            self.results.append({
                'strategy': name,
                'input': case['input'],
                'expected': case['expected'],
                'actual': response,
                'correct': case['expected'] in response
            })
        return self.summary(name)

    def summary(self, name: str) -> dict:
        subset = [r for r in self.results if r['strategy'] == name]
        accuracy = sum(r['correct'] for r in subset) / len(subset)
        return {'strategy': name, 'accuracy': accuracy, 'n': len(subset)}

    def compare_all(self):
        """输出所有策略的对比报告"""
        strategies = set(r['strategy'] for r in self.results)
        return [self.summary(s) for s in strategies]
```

```python
class LinearRegression:
    def __init__(self, lr=0.01, epochs=1000):
        self.lr = lr
        self.epochs = epochs
        self.weights = None
        self.bias = None

    def fit(self, X: np.ndarray, y: np.ndarray):
        n_samples, n_features = X.shape
        self.weights = np.zeros(n_features)
        self.bias = 0

        for epoch in range(self.epochs):
            y_pred = X @ self.weights + self.bias
            dw = (1 / n_samples) * X.T @ (y_pred - y)
            db = (1 / n_samples) * np.sum(y_pred - y)
            self.weights -= self.lr * dw
            self.bias -= self.lr * db

    def predict(self, X: np.ndarray) -> np.ndarray:
        return X @ self.weights + self.bias
```

```python
class KMeans:
    def __init__(self, k=3, max_iters=100):
        self.k = k
        self.max_iters = max_iters

    def fit(self, X):
        # 随机初始化质心

            ])
```

```python
def manual_softmax(logits):
    """手写 softmax，理解数值稳定性技巧"""
    logits_max = logits.max(dim=-1, keepdim=True).values
    exp_logits = torch.exp(logits - logits_max)
    return exp_logits / exp_logits.sum(dim=-1, keepdim=True)

def manual_cross_entropy(logits, targets):
    """手写 cross_entropy"""
    probs = manual_softmax(logits)
    n = logits.size(0)
    loss = -torch.log(probs[range(n), targets]).mean()
    return loss

class ManualLayerNorm(nn.Module):
    def __init__(self, normalized_shape, eps=1e-5):
        super().__init__()

        self.beta = nn.Parameter(torch.zeros(normalized_shape))
```

```python
import torch
import torch.nn as nn
import torch.nn.functional as F
import math

# 手写 Scaled Dot-Product Attention
class ScaledDotProductAttention(nn.Module):
    def __init__(self, dropout=0.1):
        super().__init__()
        self.dropout = nn.Dropout(dropout)

    def forward(self, Q, K, V, mask=None):
        """
        Q: (batch, heads, seq_len, d_k)
        K: (batch, heads, seq_len, d_k)
        V: (batch, heads, seq_len, d_k)
        """
        d_k = Q.size(-1)
        scores = torch.matmul(Q, K.transpose(-2, -1)) / math.sqrt(d_k)

        if mask is not None:
            scores = scores.masked_fill(mask == 0, float('-inf'))

        attn_weights = F.softmax(scores, dim=-1)
        attn_weights = self.dropout(attn_weights)
        return torch.matmul(attn_weights, V), attn_weights

# 手写 Multi-Head Attention
class MultiHeadAttention(nn.Module):
    def __init__(self, d_model, num_heads, dropout=0.1):
        super().__init__()
        assert d_model % num_heads == 0
        self.d_model = d_model
        self.num_heads = num_heads
        self.d_k = d_model // num_heads

        self.W_Q = nn.Linear(d_model, d_model)
        self.W_K = nn.Linear(d_model, d_model)
        self.W_V = nn.Linear(d_model, d_model)
        self.W_O = nn.Linear(d_model, d_model)

        self.attention = ScaledDotProductAttention(dropout)
        self.dropout = nn.Dropout(dropout)

    def forward(self, Q, K, V, mask=None):
        batch_size = Q.size(0)

        # 线性变换 + 拆头
        Q = self.W_Q(Q).view(batch_size, -1, self.num_heads, self.d_k).transpose(1, 2)
        K = self.W_K(K).view(batch_size, -1, self.num_heads, self.d_k).transpose(1, 2)
        V = self.W_V(V).view(batch_size, -1, self.num_heads, self.d_k).transpose(1, 2)

        # 计算注意力
        out, attn = self.attention(Q, K, V, mask)

        # 合并头
        out = out.transpose(1, 2).contiguous().view(batch_size, -1, self.d_model)
        return self.dropout(self.W_O(out))

# 手写 Positional Encoding（正弦编码）
class PositionalEncoding(nn.Module):
    def __init__(self, d_model, max_len=5000):
        super().__init__()
        pe = torch.zeros(max_len, d_model)
        position = torch.arange(0, max_len).unsqueeze(1).float()
        div_term = torch.exp(
            torch.arange(0, d_model, 2).float() * -(math.log(10000.0) / d_model)
        )
        pe[:, 0::2] = torch.sin(position * div_term)
        pe[:, 1::2] = torch.cos(position * div_term)
        pe = pe.unsqueeze(0)  # (1, max_len, d_model)
        self.register_buffer('pe', pe)

    def forward(self, x):
        return x + self.pe[:, :x.size(1)]

# 手写 Feed-Forward Network
class FeedForward(nn.Module):
    def __init__(self, d_model, d_ff, dropout=0.1):
        super().__init__()
        self.fc1 = nn.Linear(d_model, d_ff)
        self.fc2 = nn.Linear(d_ff, d_model)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        return self.fc2(self.dropout(F.relu(self.fc1(x))))

# 周六下午: 组装完整 Transformer Encoder
class TransformerEncoderLayer(nn.Module):
    def __init__(self, d_model, num_heads, d_ff, dropout=0.1):
        super().__init__()
        self.self_attn = MultiHeadAttention(d_model, num_heads, dropout)
        self.ffn = FeedForward(d_model, d_ff, dropout)
        self.norm1 = nn.LayerNorm(d_model)
        self.norm2 = nn.LayerNorm(d_model)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x, mask=None):
        # Self-Attention + Residual + Norm
        attn_out = self.self_attn(x, x, x, mask)
        x = self.norm1(x + self.dropout(attn_out))

        # FFN + Residual + Norm
        ffn_out = self.ffn(x)
        x = self.norm2(x + self.dropout(ffn_out))
        return x

class TransformerEncoder(nn.Module):
    def __init__(self, vocab_size, d_model, num_heads, d_ff, num_layers, dropout=0.1):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, d_model)
        self.pos_encoding = PositionalEncoding(d_model)
        self.layers = nn.ModuleList([
            TransformerEncoderLayer(d_model, num_heads, d_ff, dropout)
            for _ in range(num_layers)
        ])
        self.dropout = nn.Dropout(dropout)

    def forward(self, x, mask=None):
        x = self.embedding(x) * math.sqrt(self.embedding.embedding_dim)
        x = self.pos_encoding(x)
        x = self.dropout(x)
        for layer in self.layers:
            x = layer(x, mask)
        return x
```

```python
# GPT-2 风格 Transformer Block
class GPTBlock(nn.Module):
    def __init__(self, config):
        super().__init__()
        self.ln1 = nn.LayerNorm(config.n_embd)
        self.attn = CausalSelfAttention(config)
        self.ln2 = nn.LayerNorm(config.n_embd)
        self.mlp = nn.Sequential(
            nn.Linear(config.n_embd, 4 * config.n_embd),
            nn.GELU(),
            nn.Linear(4 * config.n_embd, config.n_embd),
            nn.Dropout(config.dropout),
        )

    def forward(self, x):
        x = x + self.attn(self.ln1(x))
        x = x + self.mlp(self.ln2(x))
        return x

class CausalSelfAttention(nn.Module):
    def __init__(self, config):
        super().__init__()
        assert config.n_embd % config.n_head == 0
        self.c_attn = nn.Linear(config.n_embd, 3 * config.n_embd)
        self.c_proj = nn.Linear(config.n_embd, config.n_embd)
        self.n_head = config.n_head
        self.n_embd = config.n_embd
        self.register_buffer(
            "mask",
            torch.tril(torch.ones(config.block_size, config.block_size))
            .view(1, 1, config.block_size, config.block_size)
        )

    def forward(self, x):
        B, T, C = x.size()  # batch, sequence, embedding_dim
        qkv = self.c_attn(x)
        q, k, v = qkv.split(C, dim=2)
        q = q.view(B, T, self.n_head, C // self.n_head).transpose(1, 2)
        k = k.view(B, T, self.n_head, C // self.n_head).transpose(1, 2)
        v = v.view(B, T, self.n_head, C // self.n_head).transpose(1, 2)

        att = (q @ k.transpose(-2, -1)) * (1.0 / math.sqrt(k.size(-1)))
        att = att.masked_fill(self.mask[:,:,:T,:T] == 0, float('-inf'))
        att = F.softmax(att, dim=-1)
        y = att @ v
        y = y.transpose(1, 2).contiguous().view(B, T, C)
        return self.c_proj(y)

class MiniGPT(nn.Module):
    def __init__(self, config):
        super().__init__()
        self.config = config
        self.transformer = nn.ModuleDict(dict(
            wte=nn.Embedding(config.vocab_size, config.n_embd),
            wpe=nn.Embedding(config.block_size, config.n_embd),
            drop=nn.Dropout(config.dropout),
            h=nn.ModuleList([GPTBlock(config) for _ in range(config.n_layer)]),
            ln_f=nn.LayerNorm(config.n_embd),
        ))
        self.lm_head = nn.Linear(config.n_embd, config.vocab_size, bias=False)
        # Weight tying
        self.transformer.wte.weight = self.lm_head.weight

    def forward(self, idx, targets=None):
        B, T = idx.shape
        pos = torch.arange(0, T, device=idx.device)
        tok_emb = self.transformer.wte(idx)
        pos_emb = self.transformer.wpe(pos)
        x = self.transformer.drop(tok_emb + pos_emb)
        for block in self.transformer.h:
            x = block(x)
        x = self.transformer.ln_f(x)
        logits = self.lm_head(x)

        if targets is None:
            return logits, None

        loss = F.cross_entropy(
            logits.view(-1, logits.size(-1)),
            targets.view(-1)
        )
        return logits, loss

    @torch.no_grad()
    def generate(self, idx, max_new_tokens, temperature=1.0, top_k=None):
        """文本生成——支持 temperature 和 top-k 采样"""
        for _ in range(max_new_tokens):
            idx_cond = idx[:, -self.config.block_size:]
            logits, _ = self(idx_cond)
            logits = logits[:, -1, :] / temperature

            if top_k is not None:
                v, _ = torch.topk(logits, min(top_k, logits.size(-1)))
                logits[logits < v[:, [-1]]] = float('-inf')

            probs = F.softmax(logits, dim=-1)
            idx_next = torch.multinomial(probs, num_samples=1)
            idx = torch.cat((idx, idx_next), dim=1)
        return idx
```

## 14. 学习节奏与防崩策略

### 14.1 6 个月的心理周期

480 小时、26 周的学习不是线性推进的。以下是典型的高危期和应对策略：

| 阶段 | 典型心理状态 | 应对策略 |
|------|-------------|---------|
| 第 3-5 周 | 新鲜感高峰，每天有明确进展 | 充分享受正反馈，多记录、多分享 |
| 第 6-8 周 | Python 工程化训练枯燥，与 AI 直接关联感降低 | 将每个练习立即应用到 AI 项目上（如用装饰器重写 ChatBot 日志），让基础语法有具体落点 |
| 第 10-12 周 | PyTorch 调试挫败感强，backward 报错频发 | PyTorch 90% 报错是维度问题。调试超过 1 小时无进展立即停止，次日带新视角再试 |
| 第 14-16 周 | Transformer 维度变化令人困惑 | 制作"维度追踪表"——每行一个操作，每列标注输入输出维度。可视化心智模型比继续阅读更有效 |
| 第 20-22 周 | Mini-GPT 训练 loss 不下降 | 先确保小数据集上能过拟合（验证代码正确性），再放大数据集 |

### 14.2 核心原则

**允许降级，不允许归零。**

| 场景 | 降级方案 |

## 16. 避坑指南

1. **碎片时间只做阅读和思考，不做编码。** 移动端不适合编程。确保白天阅读结束时能明确列出"今晚要写什么"（精确到文件/函数名）。

2. **晚上打开电脑的前 5 分钟决定效率。** 如果打开电脑时不确定要写什么，白天的碎片时间阅读就白费了。在结束碎片时间前务必规划好当晚任务。

3. **不要因为时间少而焦虑。** 480 小时高质量学习远超 700 小时低效学习。每天 2 小时专注编码 > 8 小时边刷手机边写。

4. **周六下午是不可替代的深度时间。** 这是唯一的连续 3.5 小时大块时间，用于攻克难点和综合项目。如果丢失，整周就只剩碎片时间。

5. **论文/源码需要反复阅读。** 第一遍（碎片时间）建立印象 → 第二遍（编码时对照）加深理解 → 第三遍（周末回顾）形成直觉。反复 3 次必能理解。

6. **训练耗时的解决方案：** 晚上启动训练 → 睡觉 → 次日早上通过移动端查看 wandb 日志 → 白天分析 loss 曲线 → 晚上调参再跑。利用睡眠时间完成计算。

7. **优先掌握经典论文。** 阶段五列出的 GPT-2、LLaMA、LoRA、QLoRA 四篇论文比任何前沿论文都重要 10 倍。

8. **阶段三（第 3 个月左右）是最高放弃风险期。** ML 概念多、PyTorch 调试痛苦、Transformer 第一次手写困难。此时靠的不是动力（动力最多维持 3 周），而是惯性——每天做一点，不中断。
